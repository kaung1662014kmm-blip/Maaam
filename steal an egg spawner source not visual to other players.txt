-- Steal An Egg - Advanced Spawn Script
-- Spawns eggs and pets visible to all players with base placement

local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer
local character = player.Character or player.CharacterAdded:Wait()
local humanoidRootPart = character:WaitForChild("HumanoidRootPart")
local mouse = player:GetMouse()

-- Configuration
local EGG_COLORS = {
    {name = "Gold", color = Color3.fromRGB(255, 200, 87)},
    {name = "Blue", color = Color3.fromRGB(64, 164, 223)},
    {name = "Red", color = Color3.fromRGB(255, 85, 85)},
    {name = "Green", color = Color3.fromRGB(85, 255, 127)},
    {name = "Purple", color = Color3.fromRGB(200, 100, 255)},
}

local PET_TYPES = {
    {name = "Chicken", size = Vector3.new(0.8, 0.8, 0.8), color = Color3.fromRGB(255, 200, 100)},
    {name = "Duck", size = Vector3.new(0.7, 0.7, 0.9), color = Color3.fromRGB(255, 255, 100)},
    {name = "Bunny", size = Vector3.new(0.6, 0.8, 0.6), color = Color3.fromRGB(255, 192, 203)},
}

local spawnedObjects = {}
local currentColorIndex = 1
local currentPetIndex = 1
local placementMode = false
local previewObject = nil

-- Function to create egg
local function createEgg(position, colorIndex)
    colorIndex = colorIndex or currentColorIndex
    local eggColor = EGG_COLORS[colorIndex]
    
    local egg = Instance.new("Part")
    egg.Name = "PlacedEgg_" .. tostring(tick())
    egg.Shape = Enum.PartType.Ball
    egg.Size = Vector3.new(1.2, 1.2, 1.2)
    egg.Color = eggColor.color
    egg.Material = Enum.Material.SmoothPlastic
    egg.TopSurface = Enum.SurfaceType.Smooth
    egg.BottomSurface = Enum.SurfaceType.Smooth
    egg.CanCollide = true
    egg.Anchored = false
    egg.CFrame = CFrame.new(position + Vector3.new(0, 3, 0))
    
    -- Add BillboardGui label
    local billboard = Instance.new("BillboardGui")
    billboard.Size = UDim2.new(4, 0, 2, 0)
    billboard.MaxDistance = 100
    billboard.Parent = egg
    
    local textLabel = Instance.new("TextLabel")
    textLabel.BackgroundTransparency = 1
    textLabel.TextScaled = true
    textLabel.Text = "🥚 " .. eggColor.name .. " Egg"
    textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    textLabel.Parent = billboard
    
    -- Add BodyVelocity for physics
    local bodyVelocity = Instance.new("BodyVelocity")
    bodyVelocity.Velocity = Vector3.new(0, 0, 0)
    bodyVelocity.MaxForce = Vector3.new(0, 0, 0)
    bodyVelocity.Parent = egg
    
    egg.Parent = Workspace
    table.insert(spawnedObjects, egg)
    
    print("✓ " .. eggColor.name .. " Egg spawned at: " .. tostring(position))
    return egg
end

-- Function to create pet
local function createPet(position, petIndex)
    petIndex = petIndex or currentPetIndex
    local petType = PET_TYPES[petIndex]
    
    local pet = Instance.new("Part")
    pet.Name = "PlacedPet_" .. tostring(tick())
    pet.Shape = Enum.PartType.Ball
    pet.Size = petType.size
    pet.Color = petType.color
    pet.Material = Enum.Material.SmoothPlastic
    pet.CanCollide = true
    pet.Anchored = false
    pet.CFrame = CFrame.new(position + Vector3.new(0, 3, 0))
    
    -- Add BillboardGui label
    local billboard = Instance.new("BillboardGui")
    billboard.Size = UDim2.new(4, 0, 2, 0)
    billboard.MaxDistance = 100
    billboard.Parent = pet
    
    local textLabel = Instance.new("TextLabel")
    textLabel.BackgroundTransparency = 1
    textLabel.TextScaled = true
    textLabel.Text = "🐾 " .. petType.name
    textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    textLabel.Parent = billboard
    
    pet.Parent = Workspace
    table.insert(spawnedObjects, pet)
    
    print("✓ " .. petType.name .. " spawned at: " .. tostring(position))
    return pet
end

-- Function to enter placement mode
local function enterPlacementMode(objectType)
    placementMode = true
    print("═══════════════════════════════════════")
    print("PLACEMENT MODE ACTIVE - " .. objectType:upper())
    print("Move mouse to position, click to place")
    print("Press ESC to cancel")
    print("═══════════════════════════════════════")
    
    -- Create preview object
    if objectType == "egg" then
        previewObject = Instance.new("Part")
        previewObject.Name = "EggPreview"
        previewObject.Shape = Enum.PartType.Ball
        previewObject.Size = Vector3.new(1.2, 1.2, 1.2)
        previewObject.Color = EGG_COLORS[currentColorIndex].color
    else
        previewObject = Instance.new("Part")
        previewObject.Name = "PetPreview"
        previewObject.Shape = Enum.PartType.Ball
        previewObject.Size = PET_TYPES[currentPetIndex].size
        previewObject.Color = PET_TYPES[currentPetIndex].color
    end
    
    previewObject.Material = Enum.Material.Neon
    previewObject.CanCollide = false
    previewObject.Anchored = true
    previewObject.Transparency = 0.3
    previewObject.Parent = Workspace
    
    -- Update preview position with mouse
    local connection
    connection = RunService.RenderStepped:Connect(function()
        if placementMode and previewObject then
            local hitPosition = mouse.Hit.Position
            previewObject.CFrame = CFrame.new(hitPosition + Vector3.new(0, 3, 0))
        end
    end)
    
    -- Click to place
    local mouseClickConnection
    mouseClickConnection = mouse.Button1Down:Connect(function()
        if placementMode and previewObject then
            local position = previewObject.Position
            
            if objectType == "egg" then
                createEgg(position, currentColorIndex)
            else
                createPet(position, currentPetIndex)
            end
            
            placementMode = false
            previewObject:Destroy()
            previewObject = nil
            connection:Disconnect()
            mouseClickConnection:Disconnect()
            print("✓ Object placed successfully!")
        end
    end)
end

-- Function to clear all spawned objects
local function clearAllObjects()
    for _, obj in ipairs(spawnedObjects) do
        if obj and obj.Parent then
            obj:Destroy()
        end
    end
    spawnedObjects = {}
    print("✓ All spawned objects cleared!")
end

-- Keybinds
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    
    if input.KeyCode == Enum.KeyCode.E then
        enterPlacementMode("egg")
    elseif input.KeyCode == Enum.KeyCode.P then
        enterPlacementMode("pet")
    elseif input.KeyCode == Enum.KeyCode.C then
        currentColorIndex = currentColorIndex % #EGG_COLORS + 1
        print("🎨 Egg color changed to: " .. EGG_COLORS[currentColorIndex].name)
    elseif input.KeyCode == Enum.KeyCode.T then
        currentPetIndex = currentPetIndex % #PET_TYPES + 1
        print("🐾 Pet type changed to: " .. PET_TYPES[currentPetIndex].name)
    elseif input.KeyCode == Enum.KeyCode.Escape then
        if placementMode then
            placementMode = false
            if previewObject then
                previewObject:Destroy()
                previewObject = nil
            end
            print("✗ Placement mode cancelled")
        end
    elseif input.KeyCode == Enum.KeyCode.Delete then
        clearAllObjects()
    elseif input.KeyCode == Enum.KeyCode.H then
        printHelp()
    end
end)
