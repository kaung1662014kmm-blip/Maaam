local SPEED          = 55
local AUTO_SWING     = true
local AUTO_EQUIP     = true
local AUTO_DROP_BR   = true
local HIGHLIGHT_ON   = true
local BYPASS_ON      = true

local SLAP_LIST = {"Bat","Slap","Iron Slap","Gold Slap","Diamond Slap","Emerald Slap",
                   "Ruby Slap","Dark Matter Slap","Flame Slap","Nuclear Slap",
                   "Galaxy Slap","Glitched Slap"}

local M = {
    HEIGHT        = 4.75,
    VERT_OFFSET   = 1,
    HIT_DIST      = 9,
    swingCooldown = false,

    noTargetSince = nil,
}

local Bypass = {
    VelocityCap       = 95,
    TeleportThreshold = 8,
    LastTargetPos     = nil,
    LastTargetAt      = nil,
    StableVelocity    = Vector3.zero,
    MaxAccelPerSec    = 140,
    MaxPredictOffset  = 8,
}

function Bypass.reset()
    Bypass.LastTargetPos  = nil
    Bypass.LastTargetAt   = nil
    Bypass.StableVelocity = Vector3.zero
end

function Bypass.getStableVelocity(tHRP, now)
    local raw = tHRP.AssemblyLinearVelocity
    if raw.Magnitude > Bypass.VelocityCap then
        raw = raw.Unit * Bypass.VelocityCap
    end
    local prev = Bypass.StableVelocity
    local dt   = math.max((Bypass.LastTargetAt and (now - Bypass.LastTargetAt)) or 0.016, 0.008)
    local accel = (raw - prev) / dt
    if accel.Magnitude > Bypass.MaxAccelPerSec then
        accel = accel.Unit * Bypass.MaxAccelPerSec
    end
    local stable = prev + accel * dt
    Bypass.StableVelocity = stable
    return stable
end

function Bypass.isTeleport(tHRP, now)
    local pos = tHRP.Position
    if Bypass.LastTargetPos and Bypass.LastTargetAt then
        local dt = now - Bypass.LastTargetAt
        if dt > 0 and dt < 0.5 then
            local jump = (pos - Bypass.LastTargetPos).Magnitude
            if jump > (Bypass.TeleportThreshold + Bypass.VelocityCap * dt) then
                Bypass.LastTargetPos = pos
                Bypass.LastTargetAt  = now
                return true
            end
        end
    end
    Bypass.LastTargetPos = pos
    Bypass.LastTargetAt  = now
    return false
end

function Bypass.clampPrediction(fromPos, toPos)
    local diff = toPos - fromPos
    if diff.Magnitude > Bypass.MaxPredictOffset then
        diff = diff.Unit * Bypass.MaxPredictOffset
    end
    return fromPos + diff
end

local function findBat()
    local c = player.Character
    if not c then return nil end
    local bp = player:FindFirstChildOfClass("Backpack")
    for _, ch in ipairs(c:GetChildren()) do
        if ch:IsA("Tool") and (ch.Name:lower():find("bat") or ch.Name:lower():find("slap")) then
            return ch
        end
    end
    if bp then
        for _, ch in ipairs(bp:GetChildren()) do
            if ch:IsA("Tool") and (ch.Name:lower():find("bat") or ch.Name:lower():find("slap")) then
                return ch
            end
        end
    end
    for _, name in ipairs(SLAP_LIST) do
        local t = c:FindFirstChild(name) or (bp and bp:FindFirstChild(name))
        if t then return t end
    end
    return nil
end

local function validTarget(char)
    if not char or not char.Parent then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    return hum and hrp and hum.Health > 0
end

local function getTarget(myHRP)
    if M.locked and not validTarget(M.locked) then M.locked = nil end
    if M.locked then
        return M.locked:FindFirstChild("HumanoidRootPart"), M.locked
    end
    local best, bestD = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and validTarget(p.Character) then
            local tHRP = p.Character:FindFirstChild("HumanoidRootPart")
            local d = (tHRP.Position - myHRP.Position).Magnitude
            if d < bestD then bestD, best = d, p.Character end
        end
    end
    M.locked = best
    return best and best:FindFirstChild("HumanoidRootPart"), best
end

local antiDieConns = {}
local function activateAntiDie()
    for _, c in ipairs(antiDieConns) do pcall(function() c:Disconnect() end) end
    antiDieConns = {}
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    hum.BreakJointsOnDeath = false
    hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
    table.insert(antiDieConns, hum:GetPropertyChangedSignal("Health"):Connect(function()
        if hum.Health <= 0 then hum.Health = hum.MaxHealth end
    end))
end

local function isHoldingBrainrot()
    local char = player.Character
    if not char then return false end
    for _, obj in ipairs(char:GetChildren()) do
        if obj:IsA("Tool") then
            local n = obj.Name:lower()
            if not n:find("bat") and not n:find("slap") and not n:find("medusa") then
                return true
            end
        end
    end
    return false
end

local function dropBrainrot()
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.Jump = true
        task.wait(0.1)
        hum.Jump = false
    end
end

local function swing(tool)
    if M.swingCooldown then return end
    M.swingCooldown = true
    pcall(function() tool:Activate() end)
    task.delay(0.08, function() M.swingCooldown = false end)
end

local aimbotConnection
local function startAimbot()
    if aimbotConnection then aimbotConnection:Disconnect() end
    Bypass.reset()
    activateAntiDie()

    aimbotConnection = RunService.Heartbeat:Connect(function()
        local char = player.Character
        if not char or not char.Parent then return end
        local myHRP = char:FindFirstChild("HumanoidRootPart")
        local myH   = char:FindFirstChildOfClass("Humanoid")
        if not myHRP or not myH or myH.Health <= 0 then return end

        myH.AutoRotate = false

        if AUTO_EQUIP then
            local bat = findBat()
            if bat and bat.Parent ~= char then
                pcall(function() myH:EquipTool(bat) end)
            end
        end

        if AUTO_DROP_BR and isHoldingBrainrot() then
            dropBrainrot()
        end

        myH:Move(Vector3.new(0, 0, 0), false)

        local tHRP, tChar = getTarget(myHRP)
        if tHRP and tChar and tHRP.Parent and tChar.Parent then
            M.noTargetSince = nil
            if HIGHLIGHT_ON then hl.Adornee = tChar end

            local now = tick()
            local tVel

            if BYPASS_ON then
                if Bypass.isTeleport(tHRP, now) then
                    myHRP.AssemblyLinearVelocity = Vector3.zero
                    return
                end
                tVel = Bypass.getStableVelocity(tHRP, now)
            else
                tVel = tHRP.AssemblyLinearVelocity
            end

            local predictTime = math.clamp(tVel.Magnitude / 150, 0.05, 0.2)
            local predicted = tHRP.Position + tVel * predictTime

            if BYPASS_ON then
                predicted = Bypass.clampPrediction(tHRP.Position, predicted)
            end

            local standPos = predicted + Vector3.new(0, M.HEIGHT + M.VERT_OFFSET, 0)
            local moveDir  = standPos - myHRP.Position

            local lookTarget = Vector3.new(predicted.X, myHRP.Position.Y, predicted.Z)
            if (lookTarget - myHRP.Position).Magnitude > 0.1 then
                myHRP.CFrame = CFrame.lookAt(myHRP.Position, lookTarget)
            end

            if moveDir.Magnitude > 1 then
                myHRP.AssemblyLinearVelocity = moveDir.Unit * SPEED
            else
                myHRP.AssemblyLinearVelocity = tVel
            end

            if AUTO_SWING then
                local dist = (tHRP.Position - myHRP.Position).Magnitude
                local tool = char:FindFirstChildOfClass("Tool") or findBat()
                if tool and tool.Parent == char and dist <= M.HIT_DIST then
                    swing(tool)
                end
            end
        else
            if not M.noTargetSince then M.noTargetSince = tick() end
            if tick() - M.noTargetSince > 1.5 then
                M.locked = nil
                Bypass.reset()
                if myHRP and myHRP.Parent then
                    myHRP.AssemblyLinearVelocity = Vector3.zero
                end
                if HIGHLIGHT_ON then hl.Adornee = nil end
            end
        end
    end)
end

local function stopAimbot()
    if aimbotConnection then aimbotConnection:Disconnect() aimbotConnection = nil end
    local char = player.Character
    local r   = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if r then r.AssemblyLinearVelocity = Vector3.zero end
    if hum then hum.AutoRotate = true end
    M.locked = nil
    M.noTargetSince = nil
    Bypass.reset()
    hl.Adornee = nil
end

local enabled = false
local currentKey = Enum.KeyCode.E
local waitingForKey = false

local function updateState()
    stateLabel.Text = enabled and "ON" or "OFF"
    stateLabel.TextColor3 = enabled and theme.Green or theme.Red

    local targetColor = enabled and Color3.fromRGB(30, 55, 39) or theme.Element

    TweenService:Create(
        toggle,
        TweenInfo.new(0.18),
        {BackgroundColor3 = targetColor}
    ):Play()

    TweenService:Create(
        toggleStroke,
        TweenInfo.new(0.18),
        {Color = enabled and theme.Green or theme.Border}
    ):Play()
end

local function setEnabled(value)
    enabled = value
    updateState()

    if enabled then
        startAimbot()
    else
        stopAimbot()
    end
end

toggle.Activated:Connect(function()
    setEnabled(not enabled)
end)

keyButton.Activated:Connect(function()
    waitingForKey = true
    keyButton.Text = "PRESS KEY"
    keyButton.TextColor3 = theme.Green
end)

UserInputService.InputBegan:Connect(function(input, processed)
    if processed then return end

    if waitingForKey then
        if input.UserInputType == Enum.UserInputType.Keyboard then
            if input.KeyCode == Enum.KeyCode.Escape then
                waitingForKey = false
                keyButton.Text = currentKey.Name
                keyButton.TextColor3 = theme.White
                return
            end

            currentKey = input.KeyCode
            keyButton.Text = currentKey.Name
            keyButton.TextColor3 = theme.White
            waitingForKey = false
        end
        return
    end

    if input.UserInputType == Enum.UserInputType.Keyboard
        and input.KeyCode == currentKey then
        setEnabled(not enabled)
    end
end)

local dragging = false
local dragStart
local startPosition

header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPosition = main.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if not dragging then return end

    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)
