local function protectChar(character)
    if not character then return end
    local hum = character:WaitForChild("Humanoid", 5)
    if not hum then return end
    hum.MaxHealth = math.huge
    hum.Health = math.huge
    AD.deathConns[#AD.deathConns + 1] = hum.StateChanged:Connect(function(_, new)
        if not AD.dieOn then return end
        if new == Enum.HumanoidStateType.Dead then
            hum.Health = math.huge
            hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
        end
    end)
    hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
    AD.deathConns[#AD.deathConns + 1] = hum:GetPropertyChangedSignal("Health"):Connect(function()
        if AD.dieOn and hum.Health < hum.MaxHealth then hum.Health = math.huge end
    end)
    if AD.heart then AD.heart:Disconnect() end
    AD.heart = RunService.Heartbeat:Connect(function()
        if AD.dieOn and hum and hum.Parent and hum.Health < hum.MaxHealth then hum.Health = math.huge end
    end)
end
local function startAntiDie()
    if AD.dieOn then return end
    AD.dieOn = true
    for _, c in ipairs(AD.deathConns) do pcall(function() c:Disconnect() end) end
    AD.deathConns = {}
    if AD.charConn then AD.charConn:Disconnect() end
    protectChar(LP.Character)
    AD.charConn = LP.CharacterAdded:Connect(function(c)
        if not AD.dieOn then return end
        task.wait(0.1)
        for _, c2 in ipairs(AD.deathConns) do pcall(function() c2:Disconnect() end) end
        AD.deathConns = {}
        protectChar(c)
    end)
end
local function stopAntiDie()
    if not AD.dieOn then return end
    AD.dieOn = false
    for _, c in ipairs(AD.deathConns) do pcall(function() c:Disconnect() end) end
    AD.deathConns = {}
    if AD.heart then AD.heart:Disconnect(); AD.heart = nil end
    if AD.charConn then AD.charConn:Disconnect(); AD.charConn = nil end
    local char = LP.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")

        pcall(function()
            hum:SetStateEnabled(Enum.HumanoidStateType.Dead, true)
            hum.MaxHealth = 100

end
local function startAntiDrop()
    if AD.dropOn then return end
    if type(getrawmetatable) ~= "function" or type(newcclosure) ~= "function" or type(checkcaller) ~= "function" then return end
    local ok, mt = pcall(getrawmetatable, game)
    if not ok or not mt then return end
    AD.mt, AD.oldIdx, AD.oldNew = mt, mt.__index, mt.__newindex
    pcall(setreadonly, mt, false)
    mt.__index = newcclosure(function(self, key)
        if not checkcaller() and (key == "AssemblyLinearVelocity" or key == "Velocity")
            and typeof(self) == "Instance" and self:IsA("BasePart") and self.Name == "HumanoidRootPart"
            and LP.Character and self:IsDescendantOf(LP.Character) then
            return AD.spoof
        end
        return AD.oldIdx(self, key)
    end)
    mt.__newindex = newcclosure(function(self, key, value)
        if not checkcaller() and (key == "AssemblyLinearVelocity" or key == "Velocity")
            and typeof(self) == "Instance" and self:IsA("BasePart") and self.Name == "HumanoidRootPart"
            and LP.Character and self:IsDescendantOf(LP.Character) then
            AD.spoof = value
            return
        end
        return AD.oldNew(self, key, value)
    end)
    pcall(setreadonly, mt, true)
    AD.dropOn = true
end
local function stopAntiDrop()
    if not AD.dropOn then return end
    if AD.mt and AD.oldIdx then
        pcall(setreadonly, AD.mt, false)
        AD.mt.__index = AD.oldIdx
        AD.mt.__newindex = AD.oldNew
        pcall(setreadonly, AD.mt, true)
    end
    AD.oldIdx, AD.oldNew = nil, nil
    AD.dropOn = false
end

local function clean(s) return (tostring(s):lower():gsub("[%s'%_%-]", "")) end

local function findPotion()
    local function scan(c)
        if not c then return nil end
        local exact = c:FindFirstChild("Giant Potion")
        if exact and exact:IsA("Tool") then return exact end
        for _, t in ipairs(c:GetChildren()) do
            if t:IsA("Tool") and clean(t.Name):find("giantpotion", 1, true) then return t end
        end
        return nil
    end
    return scan(LP.Character) or scan(LP:FindFirstChild("Backpack"))
end

local function usePotion(stillCarrying)
    for _ = 1, 10 do
        if stillCarrying and not stillCarrying() then return end
        local char = LP.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then
            task.wait(0.15)
        else
            local potion = findPotion()
            if not potion then return end                       
            if potion.Parent ~= char then
                pcall(function() hum:EquipTool(potion) end)
                local t0 = os.clock()
                while potion.Parent ~= char and potion.Parent ~= nil and os.clock() - t0 < 0.7 do task.wait(0.03) end
                task.wait(0.08)
            end
            if potion.Parent == char then pcall(function() potion:Activate() end) end
            local t1 = os.clock()
            while potion.Parent ~= nil and os.clock() - t1 < 0.35 do task.wait(0.05) end
            if potion.Parent == nil then return end             
        end
    end
end

local MOUNTS = { "Flying Carpet", "FlyingCarpet", "Witch's Broom", "Witch'sBroom", "WitchBroom", "Cupid's Wings", "Cupid'sWings",
    "CupidWings", "Santa's Sleigh", "Santa'sSleigh", "SantaSleigh", "Waverider" }
local function findTool(container, name)
    if not container or not name then return nil end
    local exact = container:FindFirstChild(name)
    if exact and exact:IsA("Tool") then return exact end
    local want = clean(name)
    for _, ch in ipairs(container:GetChildren()) do
        if ch:IsA("Tool") and clean(ch.Name) == want then return ch end
    end
end
local function ownedMounts()
    local out, seen = {}, {}
    local function scan(c)
        if not c then return end
        for _, t in ipairs(c:GetChildren()) do
            if t:IsA("Tool") then
                for _, m in ipairs(MOUNTS) do
                    if clean(m) == clean(t.Name) and not seen[t.Name] then seen[t.Name] = true; out[#out + 1] = t.Name end
                end
            end
        end
    end
    scan(LP:FindFirstChild("Backpack")); scan(LP.Character)
    return out
end
local function equipMount()
    local char = LP.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local function equip(name)
        local t = findTool(char, name) or findTool(LP:FindFirstChild("Backpack"), name)
        if t then
            if t.Parent ~= char then pcall(function() hum:EquipTool(t) end) end
            return true
        end
    end
    if Cfg.mount ~= "Auto" and Cfg.mount ~= "" and equip(Cfg.mount) then return end
    for _, m in ipairs(MOUNTS) do if equip(m) then return end end
end

local function makeSwitch(parent, w, h, state, onChange, offBg)
    local off = offBg or Color3.fromRGB(70, 70, 76)
    local sw = mk("TextButton", { Size = UDim2.fromOffset(w, h), BackgroundColor3 = state and GREEN or off,
        Text = "", AutoButtonColor = false, BorderSizePixel = 0 }, parent)
    corner(sw, PILL)
    local pad = math.floor(h * 0.1)
    local kd = h - pad * 2
    local knob = mk("Frame", { Size = UDim2.fromOffset(kd, kd), BackgroundColor3 = WHITE, BorderSizePixel = 0,
        Position = UDim2.fromOffset(state and (w - kd - pad) or pad, pad) }, sw)
    corner(knob, PILL)
    local cur = state and true or false
    local api = {}
    function api.set(v, instant)
        cur = v and true or false
        local pos = UDim2.fromOffset(cur and (w - kd - pad) or pad, pad)
        local col = cur and GREEN or off
        if instant then knob.Position = pos; sw.BackgroundColor3 = col
        else tween(knob, { Position = pos }); tween(sw, { BackgroundColor3 = col }) end
    end
    function api.get() return cur end
    sw.MouseButton1Click:Connect(function()
        api.set(not cur)
        if onChange then onChange(cur) end
    end)
    api.frame = sw
    return api
end

    local function paint(sel)
        for name, b in pairs(btns) do
            local on = name == sel
            tween(b.sel, { BackgroundTransparency = on and 0 or 1 })
            tween(b.lbl, { TextColor3 = on and WHITE or GREY })
        end
    end
    for idx, name in ipairs(options) do
        local b = mk("TextButton", { Size = UDim2.new(1 / n, -6, 1, -8), Position = UDim2.new((idx - 1) / n, 3 + (idx == 1 and 1 or 0), 0, 4),
            BackgroundTransparency = 1, Text = "", AutoButtonColor = false }, bar)
        local selF = mk("Frame", { Size = UDim2.fromScale(1, 1), BackgroundColor3 = Color3.fromRGB(42, 42, 47), BackgroundTransparency = 1, BorderSizePixel = 0 }, b)
        corner(selF, PILL)
        local lbl = mk("TextLabel", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = name, Font = FONT, TextSize = 17,
            TextColor3 = GREY, ZIndex = 2 }, b)
        btns[name] = { sel = selF, lbl = lbl }
        b.MouseButton1Click:Connect(function() paint(name); if onPick then onPick(name) end end)
    end
    paint(selected)
    return { paint = paint, bar = bar }
end

local alive = true
task.spawn(function()
    local last = nil
    while alive do
        local on = _G.speedBoostEnabled == true
        if on ~= last then
            last = on
            boostOn = on
            if on then startAntiDie(); startAntiDrop() else stopAntiDie(); stopAntiDrop() end
            if setBoostVisual then setBoostVisual(on) end
        end
        task.wait(0.25)
    end
end)

local function fit()
    local cam = workspace.CurrentCamera
    local vp = cam and cam.ViewportSize or Vector2.new(1280, 720)
    if IS_MOBILE then
        panelScale.Scale = math.clamp(vp.X * 0.5 / PANEL_W, 0.45, 0.9)
        winScale.Scale = math.clamp(math.min(vp.X * 0.9 / WIN_W, vp.Y * 0.88 / WIN_H), 0.45, 0.95)
    else
        panelScale.Scale = math.clamp(vp.X / 1920, 0.8, 1)
        winScale.Scale = math.clamp(math.min(vp.X / 1000, vp.Y / 600), 0.8, 1)
    end
end
fit()
do
    local cam = workspace.CurrentCamera
    if cam then track(cam:GetPropertyChangedSignal("ViewportSize"):Connect(fit)) end
end
