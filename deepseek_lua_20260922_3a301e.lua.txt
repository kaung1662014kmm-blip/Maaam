-- Sword Skins: hides the real Handle, welds a mesh replica in its place.
-- LocalScript. Client-side only, no server trust needed.

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local TextService = game:GetService("TextService")

local LP = Players.LocalPlayer

-- 1. DATA ------------------------------------------------------------
-- c0 = how the mesh sits in the hand. view = camera distance in preview.
local SKINS = {
	DiamondSword = {
		cat = "Bat", label = "Diamond Sword",
		mesh = "rbxassetid://8827558932", tex = "rbxassetid://8827558969",
		scale = Vector3.new(0.2, 0.2, 0.2),
		c0 = CFrame.new(-0.05, -0.1, -0.12) * CFrame.Angles(math.rad(90), math.rad(180), 300),
		view = 4.9,
	},
	Katana = {
		cat = "Bat", label = "Katana",
		mesh = "rbxassetid://13528902482", tex = "rbxassetid://13528902373",
		scale = Vector3.new(1.4, 1.4, 1.4),
		c0 = CFrame.new(0, 0.6, 0) * CFrame.Angles(math.rad(270), math.rad(180), math.rad(180)),
		view = 6.6,
	},
	GoldenDeagle = {
		cat = "Medusa", label = "Golden Desert Eagle",
		mesh = "rbxassetid://430251413", tex = "rbxassetid://435840335",
		scale = Vector3.new(0.01, 0.01, 0.01),
		c0 = CFrame.new(0, 0, -0.8) * CFrame.Angles(math.rad(330), math.rad(180), 0),
		view = 3.0,
	},
}

-- Which tool name belongs to which category
local function categoryOf(tool)
	local n = tool.Name:lower()
	if n:find("bat") or n:find("slap") or n:find("sword") then return "Bat" end
	if n:find("medusa") or n:find("head") then return "Medusa" end
	return nil
end

-- Chosen skin per category
local chosen = { Bat = "DiamondSword", Medusa = "GoldenDeagle" }

-- 2. BUILD / TEARDOWN -------------------------------------------------
local tracked = setmetatable({}, { __mode = "k" }) -- [tool] = model

local function buildReplica(tool, skinKey)
	local handle = tool:FindFirstChild("Handle")
	if not handle then return end

	local existing = handle:FindFirstChild("LocalReplica")
	if existing then existing:Destroy() end

	local data = SKINS[skinKey]
	if not data then return end

	-- Hide the real handle: invisible AND unhittable
	handle.Transparency = 1
	handle.CanQuery = false
	handle.CanTouch = false

	local model = Instance.new("Model")
	model.Name = "LocalReplica"

	local part = Instance.new("Part")
	part.Name = "MainPart"
	part.Size = Vector3.new(1, 1, 1)
	part.Transparency = 0
	part.CanCollide = false
	part.CanQuery = false
	part.CanTouch = false
	part.Massless = true
	part.Anchored = false
	part.Parent = model

	local mesh = Instance.new("SpecialMesh")
	mesh.MeshType = Enum.MeshType.FileMesh
	mesh.MeshId = data.mesh
	mesh.TextureId = data.tex
	mesh.Scale = data.scale
	mesh.Parent = part

	local weld = Instance.new("Motor6D")
	weld.Part0 = handle
	weld.Part1 = part
	weld.C0 = data.c0
	weld.Parent = part

	model.Parent = handle
	tracked[tool] = model

	-- If the game rewrites Transparency, put it back
	local guard
	guard = handle:GetPropertyChangedSignal("Transparency"):Connect(function()
		if not tool.Parent then guard:Disconnect() return end
		if handle.Transparency ~= 1 then handle.Transparency = 1 end
	end)
end

local function teardown(tool)
	local model = tracked[tool]
	if model then model:Destroy() end
	tracked[tool] = nil
	local handle = tool:FindFirstChild("Handle")
	if handle then
		handle.Transparency = 0
		handle.CanQuery = true
		handle.CanTouch = true
	end
end

-- 3. WATCH ------------------------------------------------------------
-- Replica is built on EQUIP (tool inside Character), not in the Backpack,
-- because the Handle only exists visually once equipped.
local function onChildAdded(child)
	if not child:IsA("Tool") then return end
	local cat = categoryOf(child)
	if not cat then return end
	buildReplica(child, chosen[cat])

	child.AncestryChanged:Connect(function(_, parent)
		if not parent then teardown(child) end
	end)

	-- Handle can be replaced by the game; rebuild when that happens
	child.ChildAdded:Connect(function(c)
		if c.Name == "Handle" then
			task.defer(function() buildReplica(child, chosen[chosen and categoryOf(child) or "Bat"]) end)
		end
	end)
end

local function hookCharacter(char)
	for _, c in ipairs(char:GetChildren()) do onChildAdded(c) end
	char.ChildAdded:Connect(onChildAdded)
end

if LP.Character then hookCharacter(LP.Character) end
LP.CharacterAdded:Connect(hookCharacter)

-- 4. PREVIEW (ViewportFrame, rotating) --------------------------------
-- Returns a ViewportFrame ready to drop into your picker UI.
function _G.makeSkinPreview(skinKey, size)
	local data = SKINS[skinKey]
	if not data then return nil end

	local vp = Instance.new("ViewportFrame")
	vp.Size = size or UDim2.new(0, 116, 0, 116)
	vp.BackgroundColor3 = Color3.fromRGB(8, 8, 11)
	vp.Ambient = Color3.fromRGB(190, 190, 200)
	vp.LightColor = Color3.fromRGB(255, 255, 255)
	vp.LightDirection = Vector3.new(-0.4, -1, -0.6)

	local part = Instance.new("Part")
	part.Anchored = true
	part.CanCollide = false
	part.Size = Vector3.new(1, 1, 1)
	part.Parent = vp

	local mesh = Instance.new("SpecialMesh")
	mesh.MeshType = Enum.MeshType.FileMesh
	mesh.MeshId = data.mesh
	mesh.TextureId = data.tex
	mesh.Scale = data.scale
	mesh.Parent = part

	local cam = Instance.new("Camera")
	cam.FieldOfView = 45
	cam.Parent = vp
	vp.CurrentCamera = cam

	local d = data.view
	cam.CFrame = CFrame.new(Vector3.new(0, d * 0.35, d), Vector3.new(0, 0, 0))

	local conn = RunService.RenderStepped:Connect(function(dt)
		if not part.Parent then conn:Disconnect() return end
		part.CFrame = part.CFrame * CFrame.Angles(0, dt * 1.2, 0)
	end)

	return vp
end

-- 5. API --------------------------------------------------------------
-- Swap the skin on a category and rebuild every equipped tool of it.
function _G.setSkin(cat, skinKey)
	if not SKINS[skinKey] or SKINS[skinKey].cat ~= cat then return end
	chosen[cat] = skinKey
	local char = LP.Character
	if not char then return end
	for _, c in ipairs(char:GetChildren()) do
		if c:IsA("Tool") and categoryOf(c) == cat then
			buildReplica(c, skinKey)
		end
	end
end
