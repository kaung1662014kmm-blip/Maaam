local function corner(inst, r)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, r or 8)
	c.Parent = inst
end

local function addStroke(inst, col, thick, trans)
	local e = Instance.new("UIStroke")
	e.Color = col or Theme.Accent
	e.Thickness = thick or 1
	e.Transparency = trans or 0.4
	e.Parent = inst
end

local function tween(inst, props, t)
	TweenService:Create(inst, TweenInfo.new(t or 0.14, Enum.EasingStyle.Quad), props):Play()
end

local function BuildAnimString(targetId)
	local Guid = "{" .. HttpService:GenerateGUID(false):upper() .. "}"
	local RandomNumbers = {}
	for _ = 1, math.random(3, 5) do
		local NumStr = ""
		for _ = 1, 15 do
			NumStr = NumStr .. tostring(math.random(0, 9))
		end
		table.insert(RandomNumbers, NumStr)
	end
	return "http" .. Guid .. "\xEF\xBC" .. table.concat(RandomNumbers, "\xEF\xBC") .. "=" .. targetId
end

local function EnforceLimbTransparency(character)
	if not FlashEnabled or not character then return end
	for _, child in ipairs(character:GetChildren()) do
		if child:IsA("BasePart") and child.Name ~= "HumanoidRootPart" then
			if child.Transparency ~= 1 then
				child.Transparency = 1
			end
		end
	end
end

local function ResetLimbTransparency(character)
	if not character then return end
	for _, child in ipairs(character:GetChildren()) do
		if child:IsA("BasePart") and child.Name ~= "HumanoidRootPart" then
			child.Transparency = 0
		end
	end
end

local function StopFreeze()
	if HeartbeatConnection then
		HeartbeatConnection:Disconnect()
		HeartbeatConnection = nil
	end
	if AnimConnection then
		AnimConnection:Disconnect()
		AnimConnection = nil
	end
	if CurrentTrack then
		CurrentTrack:Stop()
		CurrentTrack = nil
	end
end

local function ApplyFreeze(character)
	StopFreeze()
	if not FlashEnabled or not character then return end

	local Humanoid = character:WaitForChild("Humanoid", 3)
	if not Humanoid then return end
	local Animator = Humanoid:WaitForChild("Animator", 3)
	if not Animator then return end

	local AnimateScript = character:WaitForChild("Animate", 2)
	if AnimateScript then AnimateScript.Disabled = true end

	local function PlayTrack()
		if CurrentTrack then CurrentTrack:Stop() end
		local Anim = Instance.new("Animation")
		Anim.AnimationId = BuildAnimString(ANIM_ID)
		CurrentTrack = Animator:LoadAnimation(Anim)
		CurrentTrack.Priority = Enum.AnimationPriority.Action4
		CurrentTrack:Play()
		CurrentTrack.TimePosition = START_TIME
		CurrentTrack:AdjustSpeed(0)
		for _, track in ipairs(Animator:GetPlayingAnimationTracks()) do
			if track ~= CurrentTrack then track:Stop() end
		end
	end

	PlayTrack()

	HeartbeatConnection = RunService.Heartbeat:Connect(function()
		if FlashEnabled then
			if CurrentTrack and not CurrentTrack.IsPlaying then PlayTrack() end
			EnforceLimbTransparency(character)
		end
	end)

	AnimConnection = Animator.AnimationPlayed:Connect(function(track)
		if track ~= CurrentTrack then track:Stop() end
	end)
end

local sabcomTracks = {}

local function SabcomStop()
	for _, t in ipairs(sabcomTracks) do
		pcall(function() t:AdjustWeight(0, 0) end)
		pcall(function() t:Stop(0) end)
		pcall(function() t:Destroy() end)
	end
	sabcomTracks = {}
end

local function SabcomPulse()
	SabcomStop()
	local chr = LocalPlayer.Character
	if not chr then return end
	local anm = chr:FindFirstChildWhichIsA("Animator", true)
	if not anm then return end
	if not anm.LoadAnimationCoreScript then return end

	local a = Instance.new("Animation")
	a.AnimationId = "rbxassetid://" .. ANIM_ID
	local ok, t = pcall(function() return anm:LoadAnimationCoreScript(a) end)
	if not ok or not t then return end

	t:Play()
	t.Priority = Enum.AnimationPriority.Action4
	t.Looped = true
	t:AdjustSpeed(0)
	t:AdjustWeight(SABCOM_WEIGHT)
	t.TimePosition = SABCOM_TP
	sabcomTracks[1] = t

	task.wait(0.05)
	SabcomStop()
end

local donorMap
local doneChars = setmetatable({}, { __mode = "k" })
local origMeshes = setmetatable({}, { __mode = "k" })

local function GetDonorMap()
	if donorMap then return donorMap end
	donorMap = {}
	pcall(function()
		local r = Players:CreateHumanoidModelFromDescription(
			Instance.new("HumanoidDescription"), Enum.HumanoidRigType.R15)
		for _, d in ipairs(r:GetChildren()) do
			if d:IsA("MeshPart") then donorMap[d.Name] = d.MeshId end
		end
		r:Destroy()
	end)
	return donorMap
end

local function Unskin(c)
	if doneChars[c] then return end
	doneChars[c] = true
	local mp = GetDonorMap()
	for _, d in ipairs(c:GetChildren()) do
		if d:IsA("MeshPart") and mp[d.Name] then
			pcall(function()
				if d.HasSkinnedMesh then
					if origMeshes[d] == nil then origMeshes[d] = d.MeshId end
					d.MeshId = mp[d.Name]
					d.HasSkinnedMesh = false
				end
			end)
		end
	end
end

local function Reskin()
	for d, id in pairs(origMeshes) do
		pcall(function()
			d.MeshId = id
			d.HasSkinnedMesh = true
		end)
	end
	table.clear(origMeshes)
	table.clear(doneChars)
end

local function IsHuge(c)
	local ok, s = pcall(function() return c:GetExtentsSize() end)
	return ok and s and (s.X > BIG or s.Y > BIG or s.Z > BIG)
end

task.spawn(function()
	while task.wait(0.4) do
		if AntiFlashEnabled then
			for _, v in ipairs(Players:GetPlayers()) do
				if v.Character and IsHuge(v.Character) then
					pcall(Unskin, v.Character)
				end
			end
		end
	end
end)

local KillOtherAnims = true
local HookedAnimators = setmetatable({}, { __mode = "k" })

local function StopAll(animator)
	if not animator then return end
	pcall(function()
		for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
			pcall(function() track:Stop(0) end)
			pcall(function() track:AdjustSpeed(0) end)
		end
	end)
end

local function HookAnimator(animator)
	if not animator or HookedAnimators[animator] then return end
	HookedAnimators[animator] = true
	StopAll(animator)
	pcall(function()
		animator.AnimationPlayed:Connect(function(track)
			if not KillOtherAnims then return end
			pcall(function() track:Stop(0) end)
			pcall(function() track:AdjustSpeed(0) end)
		end)
	end)
end

local function HookOtherCharacter(character)
	if not character then return end
	task.spawn(function()
		local AnimateScript = character:WaitForChild("Animate", 5)
		if AnimateScript then pcall(function() AnimateScript.Disabled = true end) end
		local Humanoid = character:WaitForChild("Humanoid", 10)
		local Animator = Humanoid and Humanoid:WaitForChild("Animator", 10)
		if Animator then HookAnimator(Animator) end
	end)
end

local function HookOtherPlayer(player)
	if player == LocalPlayer then return end
	if player.Character then HookOtherCharacter(player.Character) end
	player.CharacterAdded:Connect(HookOtherCharacter)
end

for _, player in ipairs(Players:GetPlayers()) do
	pcall(HookOtherPlayer, player)
end
Players.PlayerAdded:Connect(function(player)
	pcall(HookOtherPlayer, player)
end)

task.spawn(function()
	while true do
		task.wait(0.25)
		if KillOtherAnims then
			for _, player in ipairs(Players:GetPlayers()) do
				if player ~= LocalPlayer then
					local Character = player.Character
					if Character then
						local AnimateScript = Character:FindFirstChild("Animate")
						if AnimateScript and not AnimateScript.Disabled then
							pcall(function() AnimateScript.Disabled = true end)
						end
						local Humanoid = Character:FindFirstChildOfClass("Humanoid")
						local Animator = Humanoid and Humanoid:FindFirstChildOfClass("Animator")
						if Animator then
							HookAnimator(Animator)
							StopAll(Animator)
						end
					end
				end
			end
		end
	end
end)

local togUpdaters = {}

local function MakeToggle(y, txt, key, onToggle)
	local b = Instance.new("TextButton")
	b.Position = UDim2.new(0, 8, 0, y)
	b.Size = UDim2.new(1, -16, 0, 28)
	b.BorderSizePixel = 0
	b.Font = Enum.Font.Gotham
	b.TextSize = 12
	b.TextColor3 = Theme.Text
	b.AutoButtonColor = false
	b.Parent = panel
	corner(b, 6)

	local pill = Instance.new("Frame")
	pill.Size = UDim2.new(0, 32, 0, 16)
	pill.Position = UDim2.new(1, -40, 0.5, -8)
	pill.BorderSizePixel = 0
	pill.Parent = b
	corner(pill, 8)

	local knob = Instance.new("Frame")
	knob.Size = UDim2.new(0, 12, 0, 12)
	knob.Position = UDim2.new(0, 2, 0.5, -6)
	knob.BorderSizePixel = 0
	knob.BackgroundColor3 = Theme.Text
	knob.Parent = pill
	corner(knob, 6)

	local state = false

	local function refresh()
		tween(b, { BackgroundColor3 = state and Color3.fromRGB(30, 8, 12) or Theme.Off })
		tween(pill, { BackgroundColor3 = state and Theme.On or Color3.fromRGB(48, 48, 48) })
		tween(knob, { Position = state and UDim2.new(1, -14, 0.5, -6) or UDim2.new(0, 2, 0.5, -6) })
		b.Text = "  " .. txt
		b.TextXAlignment = Enum.TextXAlignment.Left
	end

	togUpdaters[key] = function(v)
		state = v
		refresh()
	end

	b.MouseButton1Click:Connect(function()
		state = not state
		refresh()
		if onToggle then onToggle(state) end
	end)

	b.MouseEnter:Connect(function()
		tween(b, { BackgroundColor3 = state and Color3.fromRGB(38, 12, 16) or Color3.fromRGB(32, 32, 32) })
	end)
	b.MouseLeave:Connect(refresh)

	refresh()
	return function() return state end
end

		end
	end
end)
